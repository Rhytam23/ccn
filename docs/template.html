git clone <repo> && cd <repo>
python -m venv .venv && . .venv/bin/activate
pip install -r requirements.txt && pip install -e .
python scripts/run_benchmarks.py --profile full --device auto --tag my-run
python scripts/make_report.py

Or open notebooks/01_run_benchmarks_colab.ipynb in Google Colab (free T4 GPU).
